Пример кода для работодателя от кандидата: Жигарев Никита

Реализовал обработку JSON‑запросов через HTTP‑сервис 1С, по сути проверяя уникальность внешнего номера и записывая заявку только один раз.

&НаСервере
Процедура ОбработатьЗаявку()
    ТелоЗапроса = HTTPЗапрос.Тело;
    Данные = JSONИзСтроки(ТелоЗапроса);
    
    ВнешнийНомер = Данные.ВнешнийНомер;
    
    Если Не СуществуетЗаявка(ВнешнийНомер) Тогда
        // Создаём новый документ Заказ
        НовыйЗаказ = Документы.ЗаказПокупателя.Создать();
        НовыйЗаказ.Дата = ТекущаяДата();
        НовыйЗаказ.Контрагент = Справочники.Контрагенты.НайтиПоКоду(Данные.КонтрагентКод);
        // Заполняем табличную часть
        Для Каждого Строка Из Данные.Товары Цикл
            НовыйЗаказ.Товары.Добавить();
            НовыйЗаказ.Товары.Последний().Номенклатура = Справочники.Номенклатура.НайтиПоКоду(Строка.НоменклатураКод);
            НовыйЗаказ.Товары.Последний().Количество = Строка.Количество;
            НовыйЗаказ.Товары.Последний().Цена = Строка.Цена;
        КонецЦикла;
        // Пишем в журнал ошибок, если есть проблемы
        Попытка
            НовыйЗаказ.Записать();
        Исключение
            ЖурналОшибок.Записать(ОписаниеОшибки(), "ОбработкаЗаявки");
        КонецПопытки;
    Иначе
        ЖурналОшибок.Записать("Дублирование внешнего номера: " + ВнешнийНомер, "ОбработкаЗаявки");
    КонецЕсли;
КонецПроцедуры

    Запрос = Новый Запрос;
    Запрос.Текст = "
        ВЫБРАТЬ
            Заказы.Ссылка
        ИЗ
            Документ.ЗаказПокупателя.КАК Заказы

            Заказы.ВнешнийНомер = &ВнешнийНомер
    ";
    Запрос.УстановитьПараметр("ВнешнийНомер", ВнешнийНомер);
    Результат = Запрос.Выполнить();
    Возврат Результат.Выбрать().Следующий();
